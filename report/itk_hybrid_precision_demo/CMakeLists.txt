cmake_minimum_required(VERSION 3.16)
project(itk_hybrid_precision_demo LANGUAGES CXX)

find_package(ITK REQUIRED COMPONENTS ITKSmoothing ITKImageIntensity ITKThresholding ITKImageFeature ITKConvolution ITKFFT ITKImageSources ITKRegistrationMethodsv4 ITKAnisotropicSmoothing ITKIOImageBase ITKIOPNG ITKIOMeta)
include(${ITK_USE_FILE})

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(itk_hybrid_precision_demo hybrid_precision_demo.cxx)
add_executable(hybrid_precision_bench hybrid_precision_bench.cxx)
add_executable(hybrid_precision_real_bench hybrid_precision_real_bench.cxx)
add_executable(precision_pipeline_bench precision_pipeline_bench.cxx)
add_executable(precision_bilateral_bench precision_bilateral_bench.cxx)
add_executable(precision_conv_bench precision_conv_bench.cxx)
add_executable(precision_registration_bench precision_registration_bench.cxx)
add_executable(precision_diffusion_bench precision_diffusion_bench.cxx)
target_link_libraries(itk_hybrid_precision_demo PRIVATE ${ITK_LIBRARIES})
target_link_libraries(hybrid_precision_bench PRIVATE ${ITK_LIBRARIES})
target_link_libraries(hybrid_precision_real_bench PRIVATE ${ITK_LIBRARIES})
target_link_libraries(precision_pipeline_bench PRIVATE ${ITK_LIBRARIES})
target_link_libraries(precision_bilateral_bench PRIVATE ${ITK_LIBRARIES})
target_link_libraries(precision_conv_bench PRIVATE ${ITK_LIBRARIES})
target_link_libraries(precision_registration_bench PRIVATE ${ITK_LIBRARIES})
target_link_libraries(precision_diffusion_bench PRIVATE ${ITK_LIBRARIES})

if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_link_options(itk_hybrid_precision_demo PRIVATE -static-libgcc -static-libstdc++)
  target_link_options(hybrid_precision_bench PRIVATE -static-libgcc -static-libstdc++)
  target_link_options(hybrid_precision_real_bench PRIVATE -static-libgcc -static-libstdc++)
  target_link_options(precision_pipeline_bench PRIVATE -static-libgcc -static-libstdc++)
  target_link_options(precision_bilateral_bench PRIVATE -static-libgcc -static-libstdc++)
  target_link_options(precision_conv_bench PRIVATE -static-libgcc -static-libstdc++)
  target_link_options(precision_registration_bench PRIVATE -static-libgcc -static-libstdc++)
  target_link_options(precision_diffusion_bench PRIVATE -static-libgcc -static-libstdc++)
endif()
